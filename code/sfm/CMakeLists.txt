cmake_minimum_required(VERSION 3.24)
project(SeedReconstructionFinal LANGUAGES CXX)
if(NOT MSVC)
  message(FATAL_ERROR "Build with Windows x64 and MSVC.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenMP REQUIRED)
add_executable(initialize_cameras src/modules/sfm/cpp/photo_initial.cpp)
add_executable(incremental_sfm src/modules/sfm/cpp/independent_sfm.cpp)
add_executable(rebuild_tracks src/tracks/rebuild_tracks.cpp)
add_executable(maintain_tracks src/tracks/maintain_tracks.cpp)
add_executable(refine_image_observations src/tracks/refine_image_observations.cpp)
add_executable(refine_multiview_observations src/tracks/refine_image_observations.cpp)
add_executable(refine_geometry src/tracks/refine_geometry.cpp)
add_executable(refine_orbit src/tracks/refine_orbit.cpp)
add_executable(triangulate_anchors src/tracks/triangulate_anchors.cpp)
add_executable(refine_covariance src/tracks/refine_covariance.cpp)
add_executable(refine_track_segments src/tracks/refine_track_segments.cpp)
target_compile_definitions(refine_multiview_observations PRIVATE IMAGE_SHIFT_LIMIT=64)
foreach(t initialize_cameras incremental_sfm rebuild_tracks maintain_tracks refine_image_observations refine_multiview_observations refine_geometry refine_orbit triangulate_anchors refine_covariance refine_track_segments)
  target_include_directories(${t} PRIVATE src/modules/common/cpp src/modules/calibration/cpp src/modules/dense/cpp)
  target_compile_options(${t} PRIVATE /W4 /EHsc /fp:strict /utf-8)
  target_compile_definitions(${t} PRIVATE MS_USE_WIC=1)
  target_link_libraries(${t} PRIVATE bcrypt ole32 windowscodecs OpenMP::OpenMP_CXX)
endforeach()
target_compile_definitions(incremental_sfm PRIVATE MS_NATIVE_PARALLEL_COST)
enable_testing()
add_test(NAME rebuild_geometry COMMAND rebuild_tracks --self-test)
add_test(NAME maintain_geometry COMMAND maintain_tracks --self-test)
add_test(NAME image_observation_weights COMMAND refine_image_observations --self-test)
add_test(NAME multiview_observation_weights COMMAND refine_multiview_observations --self-test)
add_test(NAME geometry_refinement COMMAND refine_geometry --self-test)
add_test(NAME orbit_refinement COMMAND refine_orbit --self-test)
add_test(NAME covariance_refinement COMMAND refine_covariance --self-test)
add_test(NAME segment_refinement COMMAND refine_track_segments --self-test)
